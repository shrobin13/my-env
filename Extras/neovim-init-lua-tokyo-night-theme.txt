

-- =============================
-- TokyoNight Night Transparent

-- Set style before loading colorscheme
vim.g.tokyonight_style = "night"
vim.g.tokyonight_transparent = true
vim.g.tokyonight_hide_inactive_statusline = false

-- Load the colorscheme
vim.cmd([[colorscheme tokyonight]])

-- Make LazyVim floating windows transparent
local transparent_groups = {
  "Normal",
  "NormalNC",
  "SignColumn",
  "VertSplit",
  "StatusLine",
  "StatusLineNC",
  "LineNr",
  "CursorLineNr",
  "FloatBorder",
  "NormalFloat",
}

for _, group in ipairs(transparent_groups) do
  vim.api.nvim_set_hl(0, group, { bg = "NONE" })
end

-- Optional: make Telescope and other plugin floats transparent
local plugin_floats = {
  "TelescopeNormal",
  "TelescopeBorder",
  "WhichKeyFloat",
  "CmpDocumentation",
  "CmpPmenu",
}

for _, group in ipairs(plugin_floats) do
  vim.api.nvim_set_hl(0, group, { bg = "NONE" })
end
